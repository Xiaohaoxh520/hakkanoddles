// 修改 submitOrder 函数，实现云端同步 const submitOrder = async () => { if (!customerPhone.value.trim()) { alert('请填写取餐手机尾号！'); return; } const newOrder = { id: 'HK' + Date.now().toString().slice(-6), phone: customerPhone.value, items: [...cart.value], total: finalTotalAmount.value, payType: payType.value, status: '待核销', time: new Date().toLocaleTimeString() }; try { // 1. 先从云端获取现有订单 const getRes = await fetch('https://api.jsonbin.io/v3/b/你的BinID/latest', { headers: { 'X-Master-Key': '你的MasterKey' } }); const getData = await getRes.json(); let remoteOrders = getData.record.orders || []; // 2. 将新订单加到最前面 remoteOrders.unshift(newOrder); // 3. 提交同步保存到云端 await fetch('https://api.jsonbin.io/v3/b/你的BinID', { method: 'PUT', headers: { 'Content-Type': 'application/json', 'X-Master-Key': '你的MasterKey' }, body: JSON.stringify({ orders: remoteOrders }) }); alert('订单已提交成功！请凭取餐电话到店自提。'); cart.value = []; customerPhone.value = ''; showPayModal.value = false; } catch (error) { console.error('云端同步失败', error); alert('网络开小差了，请检查网络后重试'); } };
